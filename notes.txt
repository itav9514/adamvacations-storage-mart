     # Redirect .html → clean URL

# Main frontend routing
location / {
    try_files $uri $uri.html $uri/ /index.html;
}


    # Cache static assets for frontend (images, js, css, fonts, etc.)
    location ~* \.(jpg|jpeg|png|gif|ico|css|js|svg|woff|woff2|ttf|eot)$ {
        expires 30d;
        access_log off;
        add_header Cache-Control "public";
    }

    # Proxy for any /api/ calls (assuming your frontend calls backend API here)
    # If your frontend uses /cms/ for API calls, remove or adjust this block
    location ^~ /api/ {
        proxy_pass http://localhost:8080;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # Django CMS + Admin (mounted at /cms/)
    location ^~ /cms/ {
        proxy_pass http://unix:/run/av_cms_gunicorn.sock;  # No trailing / or :/

        # Remove include proxy_params;  ← COMMENT OUT or DELETE THIS LINE

        # Explicit headers only (no duplication)
        proxy_set_header Host $host;                    # ← This is the key one
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;

        # Critical for Django sub-path (/cms/)
        proxy_set_header SCRIPT_NAME /cms;
        proxy_set_header X-Script-Name /cms;

        proxy_http_version 1.1;
        proxy_redirect off;
    }

    # Serve Django static files directly (admin CSS/JS + your collected files)
    location ^~ /cms/static/ {
        alias /home/ec2-user/av_cms/staticfiles/;
        expires 30d;
        access_log off;
    }

    # Serve Django media/uploads
    location ^~ /cms/media/ {
        alias /home/ec2-user/av_cms/media/;
        expires 30d;
        access_log off;
    }

    # SPA fallback: Serve index.html for all non-matched frontend routes
    # (after checking static/media/cms/api)
